// Sonda do auditor_d19b_1 (fora do repositório): erro do store durante a espera do perdedor.
import { test } from 'node:test'
import assert from 'node:assert/strict'
const D = '/home/wilson-castro/Documents/projects/mira/nextjs-mfe/repos/erp-nucleo/dist'
const { identidadeDev } = await import(`${D}/adaptadores/identidade-dev.js`)
const { criarNucleoDoShell } = await import(`${D}/fabricas/criarNucleo.js`)
const { acessoFake, sessaoMemoria } = await import(`${D}/testing/index.js`)

test('SONDA: store lanca durante a espera com o token vencido -> renovarSessao rejeita (proxy prossegue), nunca ausente', async () => {
  process.env.ERP_RENOVACAO_ESPERA_MS = '300'; process.env.ERP_RENOVACAO_ESPERA_PASSO_MS = '20'
  const store = sessaoMemoria()
  let falhar = false
  const leitor = { ler: async (k) => { if (falhar) throw new Error('store fora'); return store.ler(k) } }
  const n = criarNucleoDoShell({ app: 'shell', sessao: leitor, destinos: {}, acesso: acessoFake({ modulos: [], administra: false }),
    lerCookieDeSessao: async () => undefined, escrita: { store, identidade: identidadeDev() } })
  const { url, idTransacao } = await n.sessao.iniciarLogin('/zona1')
  const u = new URL(url, 'http://shell.invalid')
  const { id } = await n.sessao.concluirLogin(idTransacao, { state: u.searchParams.get('state'), nonce: u.searchParams.get('nonce'), usuario: 'ana' })
  await store.gravar(id, { ...(await store.ler(id)), tokenExpiraEm: Date.now() - 1_000 })
  assert.ok(await store.adquirirLockRenovacao(id, 60_000))
  const p = n.sessao.renovarSessao(id)
  setTimeout(() => { falhar = true }, 50)
  let estado
  try { estado = await p } catch (e) { estado = 'lancou' }
  falhar = false
  assert.ok(await store.ler(id), 'a sessao continua no store')
  assert.notEqual(estado, 'ausente', 'erro do store na espera virou ausente: o proxy apagaria o cookie e mandaria ao login')
  assert.notEqual(estado, 'revogada')
})
